\section{The result}\label{sec:result}

\begin{theorem}[Gauss]\label{thm:gauss}
For every whole number $n \ge 1$,
\begin{equation}\label{eq:gauss}
  \sum_{i=1}^{n} i = \frac{n(n+1)}{2}.
\end{equation}
\end{theorem}

\begin{proof}
By induction. The case $n = 1$ is plain. If it holds for $n$, then
\begin{align}
  \sum_{i=1}^{n+1} i &= \frac{n(n+1)}{2} + (n+1) \\
                     &= \frac{(n+1)(n+2)}{2}. \nonumber
\end{align}
\end{proof}

\begin{lemma}\label{lem:norm}
For $x, y \in \R^n$ we have $\norm{x + y} \le \norm{x} + \norm{y}$, and
\[
  \tr\begin{pmatrix} a & b \\ c & d \end{pmatrix} = a + d .
\]
\end{lemma}

Theorem~\ref{thm:gauss} and Lemma~\ref{lem:norm} are all we need.
